local function loadCustomAsset(url, filename)
    local ok, data = pcall(function() return game:HttpGet(url) end)
    if not ok or not data or type(data) ~= "string" or #data < 100 then return nil end
    local writeOk = pcall(function() writefile(filename, data) end)
    if not writeOk then return nil end
    local assetOk, asset = pcall(function() return getcustomasset(filename) end)
    if assetOk and asset and type(asset) == "string" then return asset end
    return nil
end

local function new(class, props, parent)
    local o = Instance.new(class)
    for k, v in pairs(props) do if k ~= "Parent" then o[k] = v end end
    if parent then o.Parent = parent end
    return o
end
local function corner(p, r) return new("UICorner", {CornerRadius = UDim.new(0, r)}, p) end
local function circle(p) return new("UICorner", {CornerRadius = UDim.new(1, 0)}, p) end
local function stroke(p, c, t, tr) return new("UIStroke", {Color = c, Thickness = t, Transparency = tr or 0}, p) end
local function tween(o, t, p, s, d)
    local tw = TweenService:Create(o, TweenInfo.new(t, s or Enum.EasingStyle.Quad, d or Enum.EasingDirection.Out), p)
    tw:Play(); return tw
end
local function clamp(v, a, b) return math.clamp(v, a, b) end

RunService.RenderStepped:Connect(function(dt)
    if not window or not window.Parent then return end
    for _, a in ipairs(ambientP) do
        local pos = a.obj.Position
        local ny = pos.Y.Scale + a.vy * dt
        local nx = pos.X.Scale + math.sin(math.rad(a.phase)) * a.vx * dt
        a.phase = a.phase + 30 * dt
        if ny < -0.05 then ny = 1.05 end
        if nx < -0.05 then nx = 1.05 end
        if nx > 1.05 then nx = -0.05 end
        a.obj.Position = UDim2.new(nx, 0, ny, 0)
        a.obj.BackgroundTransparency = 0.6 + math.sin(math.rad(a.phase)) * 0.15
    end
end)

    for i = 1, 3 do
        local s = math.random(3, 5)
        local p = new("Frame", {
            Size = UDim2.new(0, s, 0, s),

        circle(p)
        edgeP[#edgeP + 1] = {
            obj = p, side = side,
            prog = (i - 1) / 3,
            speed = 0.04 + math.random(0, 20) / 1000,
        }
    end
end

RunService.RenderStepped:Connect(function(dt)
    if not window or not window.Parent then return end
    for _, e in ipairs(edgeP) do
        e.prog = e.prog + e.speed * dt
        if e.prog > 1 then e.prog = 0 end
        local s = e.obj.AbsoluteSize.X
        local pos
        if e.side == 1 then
            pos = UDim2.new(e.prog, -s/2, 0, 0)
        elseif e.side == 2 then
            pos = UDim2.new(1, -s, e.prog, -s/2)
        elseif e.side == 3 then
            pos = UDim2.new(1 - e.prog, -s/2, 1, -s)
        else
            pos = UDim2.new(0, 0, 1 - e.prog, -s/2)
        end
        e.obj.Position = pos
    end
end)

function _G.KurayamiTPCrosshair()
    local vp = cam.ViewportSize
    local ray = cam:ViewportPointToRay(vp.X/2, vp.Y/2)
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    if plr.Character then params.FilterDescendantsInstances = {plr.Character} end
    local result = workspace:Raycast(ray.Origin, ray.Direction * 5000, params)
    local hrp = plr.Character and plr.Character:FindFirstChild("HumanoidRootPart")
    if result and hrp then
        hrp.CFrame = CFrame.new(result.Position + Vector3.new(0, 3, 0))
    end
end

local tpRadiusValue = 500
local tpRingEnabled = false
local tpRingPart = nil
local _dotSize = 14

local function setRingEnabled(state)
    tpRingEnabled = state
    if state then
        if tpRingPart then tpRingPart:Destroy() end
        tpRingPart = new("Part", {
            Name = "KurayamiRadiusIndicator",
            Shape = Enum.PartType.Cylinder,
            Size = Vector3.new(0.15, tpRadiusValue * 2, tpRadiusValue * 2),
            Anchored = true, CanCollide = false, CastShadow = false,
            Material = Enum.Material.Neon,
            Color = Color3.fromRGB(255, 255, 255),
            Transparency = 0.7,
            Position = Vector3.new(0, -100, 0),
        }, workspace)
    else
        if tpRingPart then tpRingPart:Destroy(); tpRingPart = nil end
    end
end

local function setRadius(v)
    tpRadiusValue = math.floor(v)
    if tpRingPart then
        tpRingPart.Size = Vector3.new(0.15, tpRadiusValue * 2, tpRadiusValue * 2)
    end
end

task.spawn(function()
    while true do
        if tpRingEnabled and tpRingPart and tpRingPart.Parent then
            local hrp = plr.Character and plr.Character:FindFirstChild("HumanoidRootPart")
            if hrp then
                tpRingPart.CFrame = CFrame.new(hrp.Position - Vector3.new(0, 2.95, 0)) * CFrame.Angles(0, 0, math.rad(90))
            end
        end
        task.wait()
    end
end)

local TP = {}

local function getHRP(ch) return ch and ch:FindFirstChild("HumanoidRootPart") end

function TP.toPlayer(name)
    local t = Players:FindFirstChild(name)
    if not t then Notify("Error", "Player not found", "error"); return end
    local m, th = getHRP(plr.Character), getHRP(t.Character)
    if m and th then m.CFrame = th.CFrame + Vector3.new(0,3,0); Notify("TP", "Teleported to "..t.Name, "success") end
end
function TP.bringPlayer(name)
    local t = Players:FindFirstChild(name)
    if not t or not t.Character then Notify("Error", "Player not found", "error"); return end
    local m, th = getHRP(plr.Character), getHRP(t.Character)
    if m and th then th.CFrame = m.CFrame + Vector3.new(0,3,0); Notify("Bring", "Brought "..t.Name, "success") end
end
function TP.behindPlayer(name)
    local t = Players:FindFirstChild(name)
    if not t or not t.Character then Notify("Error", "Player not found", "error"); return end
    local m, th = getHRP(plr.Character), getHRP(t.Character)
    if m and th then m.CFrame = th.CFrame + th.CFrame.LookVector * -5 + Vector3.new(0,3,0); Notify("TP", "Teleported behind "..t.Name, "success") end
end
function TP.allToMe()
    local m = getHRP(plr.Character); if not m then return end
    local count = 0
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= plr and p.Character then
            local h = getHRP(p.Character)
            if h then
                local dist = (h.Position - m.Position).Magnitude
                if dist <= tpRadiusValue then
                    h.CFrame = m.CFrame + Vector3.new(0,3,0)
                    count = count + 1
                end
            end
        end
    end
    Notify("TP", "Pulled "..count.." players (radius "..tpRadiusValue..")", "success")
end
function TP.toCoords(x,y,z)
    local h = getHRP(plr.Character); if h then h.CFrame = CFrame.new(x,y,z); Notify("TP", "Teleported", "success") end
end
function TP.toMouse()
    if mouse and mouse.Hit then
        local h = getHRP(plr.Character); if h then h.CFrame = mouse.Hit + Vector3.new(0,3,0) end
    end
end

local clickTPActive = false
local toolEquip = false
local ctTool, ctGui, ctConn = nil, nil, nil
local TOOL_NAME = "Click To Horizon By Kur4yami2"

function TP.enableClickTP()
    if clickTPActive then return end
    clickTPActive = true
    ctGui = new("ScreenGui", {Name = "KurayamiClickTP", IgnoreGuiInset = true, ResetOnSpawn = false, DisplayOrder = 1200}, playerGui)
    local ind = new("Frame", {Size = UDim2.new(0,240,0,30), Position = UDim2.new(0.5,-120,0,60), BackgroundColor3 = C.ColorBg, BackgroundTransparency = 0.5, BorderSizePixel = 0, Visible = false}, ctGui)
    corner(ind, 8); stroke(ind, C.ColorBorderLight, 1, 0.5)
    new("TextLabel", {Size = UDim2.new(1,0,1,0), BackgroundTransparency = 1, Text = TOOL_NAME, TextColor3 = C.ColorTextDim, Font = Enum.Font.SourceSans, TextSize = 11}, ind)
    local bp = plr:FindFirstChildOfClass("Backpack")
    if bp then
        ctTool = new("Tool", {Name = TOOL_NAME, CanBeDropped = false, RequiresHandle = false}, bp)
        new("Part", {Name = "Handle", Transparency = 1, Size = Vector3.new(0.1,0.1,0.1), CanCollide = false, Massless = true}, ctTool)
        ctTool.GripPos = Vector3.new(0,0,0)
        ctTool.Equipped:Connect(function() toolEquip = true; ind.Visible = true end)
        ctTool.Unequipped:Connect(function() toolEquip = false; ind.Visible = false end)
    end
    ctConn = UIS.InputBegan:Connect(function(input)
        if not clickTPActive or not toolEquip then return end
        if UIS:GetFocusedTextBox() then return end
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            local gos = playerGui:GetGuiObjectsAtPosition(input.Position.X, input.Position.Y)
            for _, g in ipairs(gos) do 
                if g:IsDescendantOf(ctGui) or g:IsDescendantOf(gui) then return end
                local miniGuiRef = playerGui:FindFirstChild("KurayamiMiniTP")
                if miniGuiRef and g:IsDescendantOf(miniGuiRef) then return end
            end
            if mouse and mouse.Target then
                local h = getHRP(plr.Character); if h then h.CFrame = mouse.Hit + Vector3.new(0,3,0) end
            end
        end
    end)
end
function TP.disableClickTP()
    clickTPActive = false; toolEquip = false
    if ctConn then ctConn:Disconnect(); ctConn = nil end
    if ctTool then ctTool:Destroy(); ctTool = nil end
    if ctGui then ctGui:Destroy(); ctGui = nil end
end

local WAYPOINT_FILE = "kurayami_waypoints_" .. tostring(game.PlaceId) .. ".json"
local CONFIG_FILE   = "kurayami_configs.json"
local AUTOLOAD_FILE = "kurayami_autoload.json"

local allConfigs = {}
local currentAutoload = nil

local function persistWaypoints()
    local ok, enc = pcall(function() return HttpService:JSONEncode(savedWaypoints) end)
    if ok then pcall(function() if writefile then writefile(WAYPOINT_FILE, enc) end end) end
end

local function loadWaypoints()
    local ok, data = pcall(function() if readfile then return readfile(WAYPOINT_FILE) end; return nil end)
    if ok and data then
        local ok2, dec = pcall(function() return HttpService:JSONDecode(data) end)
        if ok2 and dec and type(dec) == "table" then
            savedWaypoints = dec
            for n, w in pairs(savedWaypoints) do
                refreshWpVisual(n, Vector3.new(w.x, w.y, w.z))
            end
        end
    end
end

local function loadAllConfigs()
    local ok, data = pcall(function() if readfile then return readfile(CONFIG_FILE) end; return nil end)
    if ok and data then
        local ok2, dec = pcall(function() return HttpService:JSONDecode(data) end)
        if ok2 and dec and type(dec) == "table" then allConfigs = dec end
    end
end

local function persistConfigs()
    local ok, enc = pcall(function() return HttpService:JSONEncode(allConfigs) end)
    if ok then pcall(function() if writefile then writefile(CONFIG_FILE, enc) end end) end
end

local function loadAutoloadSetting()
    local ok, data = pcall(function() if readfile then return readfile(AUTOLOAD_FILE) end; return nil end)
    if ok and data then
        local ok2, dec = pcall(function() return HttpService:JSONDecode(data) end)
        if ok2 and dec and type(dec) == "table" then currentAutoload = dec.name end
    end
end

local function persistAutoloadSetting()
    local enc
    if currentAutoload then enc = HttpService:JSONEncode({name = currentAutoload})
    else enc = HttpService:JSONEncode({}) end
    pcall(function() if writefile then writefile(AUTOLOAD_FILE, enc) end end)
end

local function setAutoload(name)
    currentAutoload = name
    persistAutoloadSetting()
    Notify("Config", "Auto-Load: "..name, "success")
end

local function clearAutoload()
    currentAutoload = nil
    persistAutoloadSetting()
    Notify("Config", "Auto-Load cleared", "info")
end

local function collectSettings()
    local bgc = aimDot.BackgroundColor3
    return {
        tpRadius    = tpRadiusValue,
        ringEnabled = tpRingEnabled,
        aimDot      = aimDot.Visible,
        dotSize     = _dotSize,
        dotColor    = {R = math.floor(bgc.R*255), G = math.floor(bgc.G*255), B = math.floor(bgc.B*255)},
    }
end

local function applySettings(s)
    if not s then return end
    if s.tpRadius then
        setRadius(s.tpRadius)
        if _G._kurayamiRadiusApi then
            if s.tpRadius > 1000 then _G._kurayamiRadiusApi.SetVisual(1000, s.tpRadius)
            else _G._kurayamiRadiusApi.SetVisual(s.tpRadius) end
        end
    end
    if s.ringEnabled ~= nil then
        setRingEnabled(s.ringEnabled)
        if _G._kurayamiRingToggle then _G._kurayamiRingToggle(s.ringEnabled) end
    end
    if s.aimDot ~= nil then
        aimDot.Visible = s.aimDot
        if _G._kurayamiAimDotToggle then _G._kurayamiAimDotToggle(s.aimDot) end
    end
    if s.dotSize then
        _dotSize = s.dotSize
        setDotSize(s.dotSize)
        if _G._kurayamiDotSizeApi then _G._kurayamiDotSizeApi.SetVisual(s.dotSize) end
    end
    if s.dotColor then
        setDotColor(Color3.fromRGB(s.dotColor.R, s.dotColor.G, s.dotColor.B))
    end
end

local function saveConfigNamed(name)
    if not name or name == "" then Notify("Error", "Set config name", "error"); return false end
    allConfigs[name] = { settings = collectSettings(), timestamp = os.time() }
    persistConfigs()
    Notify("Config", "Saved: "..name, "success")
    return true
end

local function loadConfigNamed(name)
    local cfg = allConfigs[name]
    if not cfg then Notify("Error", "Config not found", "error"); return end
    if cfg.settings then applySettings(cfg.settings) end
    Notify("Config", "Loaded: "..name, "success")
end

local function deleteConfigNamed(name)
    if not allConfigs[name] then return end
    allConfigs[name] = nil
    persistConfigs()
    if currentAutoload == name then clearAutoload() end
    Notify("Config", "Deleted: "..name, "warning")
end

local function getConfigList()
    local list = {}
    for k, _ in pairs(allConfigs) do table.insert(list, k) end
    table.sort(list)
    return list
end

function TP.saveWp(name, pos)
    if not name or name == "" then Notify("Error", "Set name", "error"); return end
    if not pos then local h = getHRP(plr.Character); if not h then return end; pos = h.Position end
    savedWaypoints[name] = {x = pos.X, y = pos.Y, z = pos.Z}
    refreshWpVisual(name, pos)
    persistWaypoints()
    Notify("Waypoint", "Saved: "..name, "success")
end
function TP.loadWp(name)
    local w = savedWaypoints[name]; if not w then Notify("Error", "Not found", "error"); return end
    local h = getHRP(plr.Character); if h then h.CFrame = CFrame.new(w.x,w.y,w.z); Notify("Waypoint", "Loaded", "success") end
end
function TP.deleteWp(name)
    savedWaypoints[name] = nil
    if wpFolder then local p = wpFolder:FindFirstChild(name); if p then p:Destroy() end end
    persistWaypoints()
    Notify("Waypoint", "Deleted", "warning")
end
function TP.renameWp(old, newn)
    if not savedWaypoints[old] then return end
    if savedWaypoints[newn] then Notify("Error", "Name exists", "error"); return end
    savedWaypoints[newn] = savedWaypoints[old]; savedWaypoints[old] = nil
    if wpFolder then local p = wpFolder:FindFirstChild(old); if p then p:Destroy() end end
    local w = savedWaypoints[newn]; refreshWpVisual(newn, Vector3.new(w.x,w.y,w.z))
    persistWaypoints()
    Notify("Waypoint", "Renamed", "success")
end
function TP.updateWp(name, pos)
    if not savedWaypoints[name] then return end
    savedWaypoints[name] = {x = pos.X, y = pos.Y, z = pos.Z}
    refreshWpVisual(name, pos)
    persistWaypoints()
    Notify("Waypoint", "Updated", "success")
end
function TP.toggleLoop(name)
    if loopActive then
        loopActive = false
        if loopThread then pcall(function() task.cancel(loopThread) end); loopThread = nil end
        Notify("Loop TP", "Stopped", "info")
    else
        loopActive = true
        loopThread = task.spawn(function()
            while loopActive do
                local t = Players:FindFirstChild(name)
                if t and t.Character then
                    local m, th = getHRP(plr.Character), getHRP(t.Character)
                    if m and th then m.CFrame = th.CFrame + Vector3.new(0,3,0) end
                end
                task.wait(0.3)
            end
        end)
        Notify("Loop TP", "Following", "success")
    end
end

function TP.serverHop()
    local url = "https://games.roblox.com/v1/games/"..game.PlaceId.."/servers/Public?sortOrder=Asc&limit=100"
    local ok, r = pcall(function() return HttpService:JSONDecode(game:HttpGet(url)) end)
    if ok and r and r.data then
        for _, s in ipairs(r.data) do
            if s.id ~= game.JobId and s.playing < s.maxPlayers then
                TeleportService:TeleportToPlaceInstance(game.PlaceId, s.id, plr); return
            end
        end
        Notify("Hop", "No servers", "warning")
    else
        Notify("Hop", "Failed", "error")
    end
end

local function buildSlider(parent, text, initial, minV, maxV, cb, lo)
    local row = new("Frame", {
        Size = UDim2.new(1,0,0,48), BackgroundColor3 = C.ColorBgItemHover,
        BackgroundTransparency = C.BtnTransparency, BorderSizePixel = 0, LayoutOrder = lo or 0, ZIndex = ZI,
    }, parent)
    corner(row, 6); stroke(row, C.ColorBorder, 1, 0.4)
    local lbl = new("TextLabel", {Size = UDim2.new(1,-20,0,16), Position = UDim2.new(0,10,0,4), BackgroundTransparency = 1, Text = text..": "..initial, TextColor3 = C.ColorText, Font = Enum.Font.SourceSans, TextSize = 12, TextXAlignment = Enum.TextXAlignment.Left, ZIndex = ZI+1}, row)
    local track = new("Frame", {Size = UDim2.new(1,-20,0,6), Position = UDim2.new(0,10,0,34), BackgroundColor3 = Color3.fromRGB(30,30,30), BorderSizePixel = 0, ZIndex = ZI+1}, row)
    corner(track, 3)
    local ratio = (initial - minV) / (maxV - minV)
    local fill = new("Frame", {Size = UDim2.new(ratio,0,1,0), BackgroundColor3 = C.ColorBorderBright, BorderSizePixel = 0, ZIndex = ZI+2}, track)
    corner(fill, 3)
    local dragging = false
    local currentValue = initial
    local function setVal(px)
        local rel = clamp((px - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
        local v = math.floor(minV + (maxV - minV) * rel)
        currentValue = v
        fill.Size = UDim2.new(rel,0,1,0)
        lbl.Text = text..": "..v
        if cb then cb(v) end
    end
    track.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true; setVal(i.Position.X)
        end
    end)
    UIS.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            setVal(i.Position.X)
        end
    end)
    UIS.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then dragging = false end
    end)
    local api = {}
    api.SetVisual = function(v, labelOverride)
        local clamped = clamp(v, minV, maxV)

        if open then closeList() else openList() end
    end)
    cancelBtn.MouseButton1Click:Connect(function()
        api.SetPlaceholder(placeholder or "Select...")
        closeList()
    end)
    return row, api

    for _, c in ipairs(content:GetChildren()) do if c:IsA("GuiObject") then c:Destroy() end end
end

local function buildInfoPage()
    clearContent()

    buildSection(content, "PLAYER PROFILE", 1)
    local pc = buildCard(content, 2)

    local profileRow = new("Frame", {
        Size = UDim2.new(1, 0, 0, 80),
        BackgroundTransparency = 1, LayoutOrder = 1, ZIndex = ZI,
    }, pc)

    local avatar = new("ImageLabel", {
        Size = UDim2.new(0, 72, 0, 72), Position = UDim2.new(0, 4, 0, 4),
        BackgroundColor3 = Color3.fromRGB(30,30,30), BorderSizePixel = 0,
        Image = "rbxthumb://type=AvatarHeadShot&id=" .. plr.UserId .. "&w=150&h=150",
        ZIndex = ZI + 1,
    }, profileRow)
    corner(avatar, 8); stroke(avatar, C.ColorBorderLight, 1, 0.3)

    new("TextLabel", {Size = UDim2.new(1, -92, 0, 22), Position = UDim2.new(0, 88, 0, 8),
        BackgroundTransparency = 1, Text = plr.DisplayName,
        TextColor3 = C.ColorText, Font = Enum.Font.SourceSansBold, TextSize = 16,
        TextXAlignment = Enum.TextXAlignment.Left, ZIndex = ZI+1}, profileRow)
    new("TextLabel", {Size = UDim2.new(1, -92, 0, 16), Position = UDim2.new(0, 88, 0, 32),
        BackgroundTransparency = 1, Text = "@" .. plr.Name,
        TextColor3 = C.ColorTextDim, Font = Enum.Font.SourceSans, TextSize = 12,
        TextXAlignment = Enum.TextXAlignment.Left, ZIndex = ZI+1}, profileRow)
    new("TextLabel", {Size = UDim2.new(1, -92, 0, 14), Position = UDim2.new(0, 88, 0, 50),
        BackgroundTransparency = 1, Text = "ID: " .. plr.UserId,
        TextColor3 = C.ColorTextDim, Font = Enum.Font.SourceSans, TextSize = 11,
        TextXAlignment = Enum.TextXAlignment.Left, ZIndex = ZI+1}, profileRow)

    buildSection(content, "DETAILS", 3)
    local dc = buildCard(content, 4)
    buildInfoRow(dc, "Account Age", (math.floor(plr.AccountAge) .. " days"), 1)
    buildInfoRow(dc, "Membership", plr.MembershipType.Name, 2)
    buildInfoRow(dc, "Friends", tostring(#plr:GetFriendsOnline()) .. " online", 3)

    buildSection(content, "PERFORMANCE", 5)
    local perfCard = buildCard(content, 6)
    local fpsRow = buildInfoRow(perfCard, "FPS", "0", 1)
    local pingRow = buildInfoRow(perfCard, "Ping", "0ms", 2)
    local uptimeRow = buildInfoRow(perfCard, "Uptime", "0s", 3)
    local memRow = buildInfoRow(perfCard, "Memory", "0MB", 4)

    if _G._kurayamiStatsConn then _G._kurayamiStatsConn:Disconnect() end
    local fa, ft = 0, 0
    _G._kurayamiStatsConn = RunService.RenderStepped:Connect(function(dt)
        fa = fa + 1; ft = ft + dt
        if ft >= 0.5 then
            local v = math.floor(fa/ft); fa = 0; ft = 0
            local b = fpsRow:FindFirstChildOfClass("TextBox"); if b then b.Text = tostring(v) end
            local ping = 0
            pcall(function()
                ping = math.floor(Stats.Network.ServerStatsItem["Data Ping"]:GetValue())
            end)
            local p = pingRow:FindFirstChildOfClass("TextBox"); if p then p.Text = ping .. "ms" end
            local elapsed = math.floor(tick() - (_G._kurayamiStartTime or tick()))
            local s = elapsed .. "s"
            if elapsed >= 60 then s = math.floor(elapsed/60) .. "m " .. (elapsed%60) .. "s" end
            if elapsed >= 3600 then s = math.floor(elapsed/3600) .. "h " .. math.floor((elapsed%3600)/60) .. "m" end
            local u = uptimeRow:FindFirstChildOfClass("TextBox"); if u then u.Text = s end
            local mem = math.floor(Stats:GetTotalMemoryUsageMb())
            local m = memRow:FindFirstChildOfClass("TextBox"); if m then m.Text = mem .. "MB" end
        end
    end)

    buildSection(content, "JOIN DISCORD!", 7)
    local dc2 = buildCard(content, 8)

    new("TextLabel", {
        Size = UDim2.new(1, 0, 0, 0), BackgroundTransparency = 1,
        Text = "Get updates, feature suggestions, bug reports, and more!",
        TextColor3 = C.ColorTextDim, Font = Enum.Font.SourceSans, TextSize = 11,
        TextWrapped = true, AutomaticSize = Enum.AutomaticSize.Y,
        TextXAlignment = Enum.TextXAlignment.Left, LayoutOrder = 1, ZIndex = ZI,
    }, dc2)

    local rDc = buildRow(dc2, 2)
    buildButton(rDc, "Join Discord!", function()
        pcall(function()
            game:GetService("GuiService"):OpenBrowserWindow("https://discord.gg/placeholder")
        end)
        Notify("Discord", "Opening in browser...", "info", 1.5)
    end, 1, UDim2.new(0.5, -3, 1, 0))
    buildButton(rDc, "Copy Link", function()
        if setclipboard then
            pcall(function() setclipboard("https://discord.gg/placeholder") end)
            Notify("Discord", "Link copied", "success", 1.5)
        end
    end, 2, UDim2.new(0.5, -3, 1, 0))
end

local function buildTeleportPage()
    clearContent()

    buildSection(content, "PLAYER TELEPORT", 1)
    local c1 = buildCard(content, 2)

    local _, pdd = buildDropdown(c1, "Select player...", 1)
    pdd.OnSelect(function(n) selPlayer = n end)

    if _G._kurayamiPlayerConns then
        for _, c in ipairs(_G._kurayamiPlayerConns) do pcall(function() c:Disconnect() end) end
    end
    _G._kurayamiPlayerConns = {}

    local function populate()
        local list = {}
        for _, p in ipairs(Players:GetPlayers()) do if p ~= plr then table.insert(list, p.Name) end end
        table.sort(list); pdd.SetOptions(list)
    end
    populate()
    table.insert(_G._kurayamiPlayerConns, Players.PlayerAdded:Connect(populate))
    table.insert(_G._kurayamiPlayerConns, Players.PlayerRemoving:Connect(function() task.wait(0.3); populate() end))

    local r1 = buildRow(c1, 2)
    buildButton(r1, "Confirm Teleport", actTPToPlayer, 1, UDim2.new(0.5,-3,1,0))
    buildButton(r1, "Refresh List", function() populate(); Notify("Players", "Refreshed", "info", 1.2) end, 2, UDim2.new(0.5,-3,1,0))

    local r2 = buildRow(c1, 3)
    buildButton(r2, "Bring Player", actBring, 1, UDim2.new(0.5,-3,1,0))
    buildButton(r2, "TP Behind", actBehind, 2, UDim2.new(0.5,-3,1,0))

    local r3 = buildRow(c1, 4)
    buildButton(r3, "TP All to Me", actAllToMe, 1, UDim2.new(0.5,-3,1,0))
    local lc = new("Frame", {Size = UDim2.new(0.5,-3,1,0), BackgroundTransparency = 1, LayoutOrder = 2, ZIndex = ZI}, r3)
    local lt = buildToggle(lc, "Loop TP", false, actLoop, 1)
    lt.Size = UDim2.new(1,0,1,0)

    local radiusSlider, radiusApi = buildSlider(c1, "TP Radius", tpRadiusValue, 0, 1000, function(v)
        setRadius(v)
    end, 5)
    _G._kurayamiRadiusApi = radiusApi

    local rRadius = buildRow(c1, 6)
    local radiusCustomBox = buildTextbox(rRadius, "Custom radius...", "", function(t) _G._kurayamiRadiusBuf = t end, 1)
    radiusCustomBox.Size = UDim2.new(0.6, -3, 1, 0)
    buildButton(rRadius, "Apply", function()
        local v = tonumber(_G._kurayamiRadiusBuf)
        if not v or v <= 0 then Notify("Error", "Invalid radius", "error"); return end
        v = math.floor(v)
        setRadius(v)
        if v > 1000 then radiusApi.SetVisual(1000, v)
        else radiusApi.SetVisual(v) end
        Notify("Radius", "Set to "..v, "success", 1.5)
    end, 2, UDim2.new(0.4, -3, 1, 0))

    local ringToggleRow, ringToggleApi = buildToggle(c1, "Show Radius Ring", false, function(s) setRingEnabled(s) end, 7)
    _G._kurayamiRingToggle = function(v) if ringToggleApi then ringToggleApi.SetState(v) end end

    local rOpen = buildRow(c1, 8)
    buildButton(rOpen, "Open Mini TP Menu", function()
        if miniGui and miniGui.Parent then
            miniHolder.Visible = true
        end
    end, 1, UDim2.new(1, 0, 1, 0))

    buildSection(content, "COORDINATE TELEPORT", 3)
    local c2 = buildCard(content, 4)
    local cbox = buildTextbox(c2, "X,Y,Z (ex: 100,50,200)", "", function(t) curCoords = t end, 1)
    local r4 = buildRow(c2, 2)
    buildButton(r4, "TP to Coords", function()
        local pts = {}
        for n in string.gmatch(curCoords, "([^,]+)") do table.insert(pts, tonumber(n)) end
        if #pts == 3 then
            ShowConfirm("Teleport", "Teleport to ("..pts[1]..", "..pts[2]..", "..pts[3]..")?", function() TP.toCoords(pts[1],pts[2],pts[3]) end)
        else Notify("Error", "Invalid coords", "error") end
    end, 1, UDim2.new(0.5,-3,1,0))
    buildButton(r4, "Copy My Pos", function()
        local h = getHRP(plr.Character)
        if h then
            local p = h.Position; local s = string.format("%d,%d,%d", p.X, p.Y, p.Z)
            curCoords = s; cbox.Text = s
            if setclipboard then pcall(function() setclipboard(s) end) end
            Notify("Copy", s, "info")
        end
    end, 2, UDim2.new(0.5,-3,1,0))

    buildSection(content, "MOUSE TELEPORT", 5)
    local c3 = buildCard(content, 6)
    local r5 = buildRow(c3, 1)
    buildButton(r5, "TP to Mouse", function() ShowConfirm("Teleport", "Teleport to mouse?", function() TP.toMouse() end) end, 1, UDim2.new(0.5,-3,1,0))
    local ctC = new("Frame", {Size = UDim2.new(0.5,-3,1,0), BackgroundTransparency = 1, LayoutOrder = 2, ZIndex = ZI}, r5)
    local ctT = buildToggle(ctC, "Click TP", false, function(s) if s then TP.enableClickTP() else TP.disableClickTP() end end, 1)
    ctT.Size = UDim2.new(1,0,1,0)

    local aimCard = buildCard(c3, 2)
    local aimDotRow, aimDotToggleApi = buildToggle(aimCard, "Show Aim Dot", false, function(s) aimDot.Visible = s end, 1)
    _G._kurayamiAimDotToggle = function(v) if aimDotToggleApi then aimDotToggleApi.SetState(v) end end

    local dotSliderRow, dotSliderApi = buildSlider(aimCard, "Dot Size", 14, 6, 40, function(v) _dotSize = v; setDotSize(v) end, 2)
    _G._kurayamiDotSizeApi = dotSliderApi

    new("TextLabel", {Size = UDim2.new(1,0,0,16), BackgroundTransparency = 1, Text = "Dot Color", TextColor3 = C.ColorTextDim, Font = Enum.Font.SourceSans, TextSize = 11, TextXAlignment = Enum.TextXAlignment.Left, LayoutOrder = 3, ZIndex = ZI}, aimCard)
    local colorRow = buildRow(aimCard, 4)
    local colors = {
        {name = "W", c = Color3.fromRGB(255,255,255)},
        {name = "R", c = Color3.fromRGB(255,80,80)},
        {name = "G", c = Color3.fromRGB(80,255,120)},
        {name = "B", c = Color3.fromRGB(80,180,255)},
    }
    for i, cl in ipairs(colors) do
        local b = new("TextButton", {
            Size = UDim2.new(0,32,1,0), BackgroundColor3 = cl.c,
            Text = cl.name, TextColor3 = Color3.fromRGB(0,0,0),
            Font = Enum.Font.SourceSansBold, TextSize = 11, BorderSizePixel = 0,
            AutoButtonColor = false, LayoutOrder = i, ZIndex = ZI,
        }, colorRow)
        corner(b, 6)
        b.MouseButton1Click:Connect(function() setDotColor(cl.c) end)
    end

    buildSection(content, "WAYPOINT", 7)
    local c4 = buildCard(content, 8)
    buildTextbox(c4, "Waypoint name...", "", function(t) wpNameBuf = t end, 1)

    local rWp1 = buildRow(c4, 2)
    buildButton(rWp1, "Save Here", function()
        if wpNameBuf == "" then Notify("Error", "Set name", "error"); return end
        TP.saveWp(wpNameBuf, nil)
        if _G._kurayamiRefreshWpList then _G._kurayamiRefreshWpList() end
    end, 1, UDim2.new(0.5,-3,1,0))
    buildButton(rWp1, "Save From Coords", function()
        if wpNameBuf == "" then Notify("Error", "Set name", "error"); return end
        if curCoords == "" then Notify("Error", "Set coords first", "error"); return end
        local pts = {}
        for n in string.gmatch(curCoords, "([^,]+)") do table.insert(pts, tonumber(n)) end
        if #pts == 3 then
            TP.saveWp(wpNameBuf, Vector3.new(pts[1],pts[2],pts[3]))
            if _G._kurayamiRefreshWpList then _G._kurayamiRefreshWpList() end
        else Notify("Error", "Invalid coords", "error") end
    end, 2, UDim2.new(0.5,-3,1,0))

    local _, wpdd = buildDropdown(c4, "Select waypoint...", 3)
    wpdd.OnSelect(function(n) selWP = n end)

    _G._kurayamiRefreshWpList = function()
        local list = {}
        for n, _ in pairs(savedWaypoints) do table.insert(list, n) end
        table.sort(list); wpdd.SetOptions(list)
    end
    _G._kurayamiRefreshWpList()

    local rWp2 = buildRow(c4, 4)
    buildButton(rWp2, "Load", function()
        if not selWP then Notify("Error", "Select waypoint", "error"); return end
        local n = selWP
        ShowConfirm("Waypoint", "Teleport to ["..n.."]?", function() TP.loadWp(n) end)
    end, 1, UDim2.new(0.5,-3,1,0))
    buildButton(rWp2, "Delete", function()
        if not selWP then Notify("Error", "Select waypoint", "error"); return end
        local n = selWP
        ShowConfirm("Delete", "Delete ["..n.."]?", function()
            TP.deleteWp(n); selWP = nil; wpdd.SetPlaceholder("Select waypoint...")
            if _G._kurayamiRefreshWpList then _G._kurayamiRefreshWpList() end
        end)
    end, 2, UDim2.new(0.5,-3,1,0))

    local rWp3 = buildRow(c4, 5)
    buildButton(rWp3, "Rename", function()
        if not selWP then Notify("Error", "Select waypoint", "error"); return end
        if wpNameBuf == "" then Notify("Error", "Set new name", "error"); return end
        TP.renameWp(selWP, wpNameBuf); selWP = wpNameBuf
        if _G._kurayamiRefreshWpList then _G._kurayamiRefreshWpList() end
    end, 1, UDim2.new(0.5,-3,1,0))
    buildButton(rWp3, "Update Pos", function()
        if not selWP then Notify("Error", "Select waypoint", "error"); return end
        local n = selWP
        ShowConfirm("Update", "Update position of ["..n.."]?", function()
            local h = getHRP(plr.Character); if h then TP.updateWp(n, h.Position) end
            if _G._kurayamiRefreshWpList then _G._kurayamiRefreshWpList() end
        end)
    end, 2, UDim2.new(0.5,-3,1,0))

    local rWp4 = buildRow(c4, 6)
    buildButton(rWp4, "Refresh List", function()
        if _G._kurayamiRefreshWpList then _G._kurayamiRefreshWpList() end
        Notify("Waypoint", "Refreshed", "info", 1.2)
    end, 1, UDim2.new(1,0,1,0))
end

    local r2 = buildRow(c1, 4)
    buildButton(r2, "Load Config", function()
        if not selConfig then Notify("Error", "Select config", "error"); return end
        local n = selConfig
        ShowConfirm("Load Config", "Load config ["..n.."]?", function() loadConfigNamed(n) end)
    end, 1, UDim2.new(0.5, -3, 1, 0))
    buildButton(r2, "Delete Config", function()
        if not selConfig then Notify("Error", "Select config", "error"); return end
        local n = selConfig
        ShowConfirm("Delete Config", "Delete config ["..n.."]?", function()
            deleteConfigNamed(n)
            selConfig = nil
            cdd.SetPlaceholder("Select config...")
            if _G._kurayamiRefreshConfigList then _G._kurayamiRefreshConfigList() end
            autoloadLbl.Text = "Current: " .. (currentAutoload or "none")
            autoloadLbl.TextColor3 = currentAutoload and C.ColorGreen or C.ColorTextDim
        end)
    end, 2, UDim2.new(0.5, -3, 1, 0))

    buildSection(content, "AUTO-LOAD CONTROL", 5)
    local c2 = buildCard(content, 6)
    local r3 = buildRow(c2, 1)
    buildButton(r3, "Set Auto-Load", function()
        if not selConfig then Notify("Error", "Select config", "error"); return end
        setAutoload(selConfig)
        autoloadLbl.Text = "Current: " .. selConfig
        autoloadLbl.TextColor3 = C.ColorGreen
    end, 1, UDim2.new(0.5, -3, 1, 0))
    buildButton(r3, "Remove Auto-Load", function()
        if not currentAutoload then Notify("Error", "No auto-load set", "error"); return end

        autoloadLbl.Text = "Current: none"

    buildSection(content, "DANGER ZONE", 7)
    local c3 = buildCard(content, 8)
    local unloadBtn = buildButton(c3, "Unload Script", actUnload, 1)
    unloadBtn.TextColor3 = C.ColorRed

local isOpen = false
local isCollapsed = false
local fullSize = UDim2.new(C.WindowScaleX, 0, C.WindowScaleY, 0)
local collapsedSize = UDim2.new(C.WindowScaleX, 0, 0, C.HeaderHeight + C.BottomBarHeight)

local function updateTglBtn()
    tween(btnTgl, 0.15, {Rotation = isCollapsed and 45 or 0}, Enum.EasingStyle.Quad)
end

local function expandWindow()
    if not isCollapsed then return end
    isCollapsed = false
    holder.Position = UDim2.new(0.5,0,0.5,0)
    tween(holder, 0.2, {Size = fullSize}, Enum.EasingStyle.Back)
    updateTglBtn()
end
local function collapseWindow()
    if isCollapsed then return end
    isCollapsed = true
    tween(holder, 0.2, {Size = collapsedSize}, Enum.EasingStyle.Back)
    updateTglBtn()
end

btnTgl.MouseButton1Click:Connect(function()
    if isCollapsed then expandWindow() else collapseWindow() end
end)

local function openWindow()
    if isOpen then return end
    isOpen = true; holder.Visible = true
    holderScale.Scale = 0.25
    tween(holderScale, 0.25, {Scale = 1}, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
end
local function closeWindow()
    if not isOpen then return end
    isOpen = false
    tween(holderScale, 0.15, {Scale = 0.25})
    task.wait(0.18); if not isOpen then holder.Visible = false end
end

local function makeDraggable(dObj)
    local dg, ds, sp
    dObj.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dg = true; ds = i.Position; sp = holder.Position
        end
    end)
    UIS.InputChanged:Connect(function(i)
        if dg and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            local d = i.Position - ds
            local nx = sp.X.Offset + d.X; local ny = sp.Y.Offset + d.Y
            local vp = workspace.CurrentCamera.ViewportSize
            nx = clamp(nx, -vp.X/2 + 40, vp.X/2 - 40)
            ny = clamp(ny, -vp.Y/2 + 40, vp.Y/2 - 40)
            holder.Position = UDim2.new(sp.X.Scale, nx, sp.Y.Scale, ny)
        end
    end)
    UIS.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then dg = false end
    end)
end
makeDraggable(header)
makeDraggable(bottomBar)

do
    local dg, mv, ds, sp
    toggle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dg = true; mv = false; ds = i.Position; sp = toggle.Position
        end
    end)
    UIS.InputChanged:Connect(function(i)
        if dg and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            local d = i.Position - ds
            if math.abs(d.X) + math.abs(d.Y) > 8 then mv = true end
            if mv then
                local np = UDim2.new(sp.X.Scale, sp.X.Offset + d.X, sp.Y.Scale, sp.Y.Offset + d.Y)
                toggle.Position = np; toggleRing.Position = np
            end
        end
    end)
    UIS.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            if dg and not mv then
                if isOpen then closeWindow() else openWindow() end
            end
            dg = false; mv = false
        end
    end)
end
